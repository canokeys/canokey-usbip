cmake_minimum_required(VERSION 3.13)
project(canokey-usbip C)

option(ENABLE_TESTS "Build for tests" OFF)
option(CANOKEY_CORE_DEBUG_OUTPUT "Enable verbose canokey-core debug output (may contain sensitive test material)" OFF)
set(CANOKEY_CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/canokey-core" CACHE PATH
        "Path to the canokey-core source tree")
set(CANOKEY_CORE_SHA "unknown" CACHE STRING
        "canokey-core revision embedded in the virtual device")
set(CANOKEY_FIRMWARE_VERSION "" CACHE STRING
        "CanoKey firmware version reported by the virtual admin applet")

if (CANOKEY_FIRMWARE_VERSION STREQUAL "")
        message(FATAL_ERROR "CANOKEY_FIRMWARE_VERSION is required")
endif ()

get_filename_component(CANOKEY_CORE_DIR "${CANOKEY_CORE_DIR}" ABSOLUTE)
if (NOT EXISTS "${CANOKEY_CORE_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "CANOKEY_CORE_DIR is not a canokey-core source tree: ${CANOKEY_CORE_DIR}")
endif ()

execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}" rev-parse HEAD
        OUTPUT_VARIABLE CANOKEY_USBIP_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if (CANOKEY_CORE_SHA STREQUAL "unknown")
        execute_process(COMMAND git -C "${CANOKEY_CORE_DIR}" rev-parse HEAD
                OUTPUT_VARIABLE CANOKEY_CORE_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
        if (CANOKEY_CORE_SHA STREQUAL "")
                set(CANOKEY_CORE_SHA "unknown")
        endif ()
endif ()
if (CANOKEY_USBIP_SHA STREQUAL "")
        set(CANOKEY_USBIP_SHA "unknown")
endif ()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall")
set(CMAKE_C_FLAGS_DEBUG "${CMAKE_C_FLAGS_DEBUG} -fno-omit-frame-pointer")

# Old canokey-core releases vendor an mbedTLS CMake file whose policy floor is
# rejected by recent CMake versions. This keeps those releases buildable while
# leaving the caller's selected core tree untouched.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "Minimum compatibility policy for vendored dependencies")
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "Do not turn vendored mbedTLS warnings into errors" FORCE)

message(STATUS "canokey-usbip SHA: ${CANOKEY_USBIP_SHA}")
message(STATUS "canokey-core SHA: ${CANOKEY_CORE_SHA}")
message(STATUS "CanoKey firmware version: ${CANOKEY_FIRMWARE_VERSION}")
message(STATUS "C compiler: ${CMAKE_C_COMPILER_ID} ${CMAKE_C_COMPILER_VERSION} (${CMAKE_C_COMPILER})")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/canokey-build-metadata.json"
        "{\n  \"canokey_usbip_sha\": \"${CANOKEY_USBIP_SHA}\",\n"
        "  \"canokey_core_sha\": \"${CANOKEY_CORE_SHA}\",\n"
        "  \"canokey_firmware_version\": \"${CANOKEY_FIRMWARE_VERSION}\",\n"
        "  \"compiler_id\": \"${CMAKE_C_COMPILER_ID}\",\n"
        "  \"compiler_version\": \"${CMAKE_C_COMPILER_VERSION}\"\n}\n")

set(VIRTCARD ON CACHE BOOL "Enable VIRTCARD in core" FORCE)
set(ENABLE_DEBUG_OUTPUT ${CANOKEY_CORE_DEBUG_OUTPUT} CACHE BOOL "canokey-core debug output" FORCE)
# USB/IP currently exports CTAP HID, WebUSB, and CCID. Keep the optional
# keyboard interface disabled so the host-side endpoint model stays stable.
set(ENABLE_IFACE_KBDHID OFF CACHE BOOL "Disable keyboard HID in USB/IP builds" FORCE)
set(USBD_PRODUCT_STRING "CanoKey USB/IP")
set(PRODUCT_NAME "CanoKey USB/IP")
add_subdirectory("${CANOKEY_CORE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/canokey-core" EXCLUDE_FROM_ALL)

file(GLOB_RECURSE SRC Src/*.c)

# device-sim.c includes a generated git-rev.h from its own directory. Build a
# private copy so configuring against an external core never writes into it.
set(COMPAT_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/compat-core")
file(MAKE_DIRECTORY "${COMPAT_GENERATED_DIR}")
configure_file("${CANOKEY_CORE_DIR}/virt-card/device-sim.c"
        "${COMPAT_GENERATED_DIR}/device-sim.c" COPYONLY)
file(WRITE "${COMPAT_GENERATED_DIR}/git-rev.h"
        "/* Generated by canokey-usbip. */\n#define GIT_REV \"${CANOKEY_FIRMWARE_VERSION}\"\n")
file(WRITE "${COMPAT_GENERATED_DIR}/virt-card-git-rev.h"
        "/* Generated by canokey-usbip. */\n#define GIT_REV \"${CANOKEY_FIRMWARE_VERSION}\"\n")

add_executable(canokey-usbip ${SRC}
        "${COMPAT_GENERATED_DIR}/device-sim.c"
        "${CANOKEY_CORE_DIR}/virt-card/fabrication.c"
        "${CANOKEY_CORE_DIR}/littlefs/bd/lfs_filebd.c")
target_include_directories(canokey-usbip SYSTEM PRIVATE
        "${COMPAT_GENERATED_DIR}"
        "${CANOKEY_CORE_DIR}/virt-card"
        "${CANOKEY_CORE_DIR}")
file(READ "${CANOKEY_CORE_DIR}/include/device.h" CANOKEY_DEVICE_HEADER)
if (CANOKEY_DEVICE_HEADER MATCHES "void[ \t\r\n]+device_loop[ \t\r\n]*\\([ \t\r\n]*void[ \t\r\n]*\\)")
        target_compile_definitions(canokey-usbip PRIVATE CANOKEY_DEVICE_LOOP_NO_ARGS=1)
endif ()
file(READ "${CANOKEY_CORE_DIR}/interfaces/USB/device/usb_device.h" CANOKEY_USB_DEVICE_HEADER)
if (CANOKEY_USB_DEVICE_HEADER MATCHES "EP_NUM_ctap_hid")
        target_compile_definitions(canokey-usbip PRIVATE CANOKEY_USB_STATIC_LAYOUT=1)
endif ()
file(READ "${CANOKEY_CORE_DIR}/interfaces/USB/core/inc/usbd_core.h" CANOKEY_USBD_CORE_HEADER)
if (CANOKEY_USBD_CORE_HEADER MATCHES "USBD_LL_Init_Done")
        target_compile_definitions(canokey-usbip PRIVATE CANOKEY_USBD_LL_INIT_DONE=1)
endif ()
file(READ "${CANOKEY_CORE_DIR}/interfaces/USB/core/inc/usbd_def.h" CANOKEY_USBD_DEF_HEADER)
if (CANOKEY_USBD_DEF_HEADER MATCHES "USBD_EndpointTypeDef[ \t\r\n]+ep0_in")
        target_compile_definitions(canokey-usbip PRIVATE CANOKEY_USBD_COMPACT_STATE=1)
endif ()
if (ENABLE_TESTS)
        target_compile_options(canokey-usbip PRIVATE "--coverage" "-fsanitize=address" "-fsanitize=undefined")
        target_link_options(canokey-usbip PRIVATE "--coverage" "-fsanitize=address" "-fsanitize=undefined")
endif ()
target_compile_definitions(canokey-usbip PRIVATE
        HW_VARIANT_NAME="CanoKey USB/IP"
        VIRT_VENDOR)
target_link_libraries(canokey-usbip canokey-core)
